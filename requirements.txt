# requirements.txt — Raspberry Pi runtime dependencies
# Install on the Pi with: pip install -r requirements.txt
#
# System packages (installed via apt in setup.sh — do NOT pip-install these):
#   python3-opencv  — cv2 + cv2.dnn for YOLO inference (ARM-compiled, faster than pip wheel)
#   python3-numpy   — array operations
#   python3-picamera2, python3-lgpio — camera and GPIO

# ── Pi runtime ──────────────────────────────────────────────────────────────

rpi-lgpio
lgpio

# WebSocket server
websockets

# WebRTC (camera streaming)
aiohttp==3.10.*
aiortc>=1.14.0

# Config file parsing
pyyaml

# ── Development / model export (Mac only, not needed on Pi) ─────────────────
#
# Export yolov8n to ONNX once on your Mac, then scp the result to the Pi:
#
#   pip install ultralytics onnx
#   python3 -c "from ultralytics import YOLO; YOLO('yolov8n.pt').export(format='onnx', imgsz=320)"
#   scp yolov8n.onnx pi@<pi-ip>:~/robo-pi/src/components/ai/models/yolov8n_320.onnx
#
# Facial tracking uses YuNet (cv2.FaceDetectorYN), a prebuilt ~230 KB ONNX — no
# export step, just download and copy. Needs OpenCV >= 4.7 for the 2023mar model:
#
#   curl -L -o face_detection_yunet.onnx \
#     https://github.com/opencv/opencv_zoo/raw/main/models/face_detection_yunet/face_detection_yunet_2023mar.onnx
#   scp face_detection_yunet.onnx pi@<pi-ip>:~/robo-pi/src/components/ai/models/face_detection_yunet.onnx
#
# Both model files are gitignored (src/components/ai/models/*). setup.sh creates
# the directory. Facial tracking falls back to Haar cascades if YuNet is absent.
